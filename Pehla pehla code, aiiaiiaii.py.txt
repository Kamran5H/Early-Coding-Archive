import numpy as np
from sklearn.linear_model import LinearRegression

# Example data: [Molecular Weight, Boiling Point (C)]
data = np.array([
    [18, 100],   # water
    [46, 78],    # ethanol
    [16, -161],  # methane
    [78, 80],    # benzene
    [58, 56]     # acetone
])

X = data[:, 0].reshape(-1, 1)
y = data[:, 1]

model = LinearRegression()
model.fit(X, y)

propanol_mw = np.array([[60]])
predicted_bp = model.predict(propanol_mw)
print(f"Predicted boiling point for MW=60: {predicted_bp[0]:.2f} °C")
